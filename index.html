3. Count Vowels and Consonants

# Implement a program that accepts a string and counts the number of vowels and consonants

# Enter a string
text = input("Enter a string: ")

# Convert the string to lower case
text = text.lower()

# Initialize counters
vowel_count = 0
consonant_count = 0

# Define vowels
vowels = "aeiou"

# Count vowels and consonants
for char in text:
    if char.isalpha():
        if char in vowels:
            vowel_count += 1
        else:
            consonant_count += 1

# Display the results
print("Number of vowels:", vowel_count)
print("Number of consonants:", consonant_count)

7. Pandas Series (Indexing, Slicing, Querying)

import pandas as pd

# Create pandas Series
marks = pd.Series(
    [85, 90, 78, 92, 88],
    index=["abhishek", "astha", "rohan", "maya", "mohan"]
)

print("Student marks series:\n", marks)

# Indexing
print("Marks of Astha:", marks["astha"])
print("Mark at 2nd position (by position):", marks.iloc[2])

# Slicing
print("\nSlicing from Astha to Maya:\n", marks["astha":"maya"])
print("\nSlicing first 3 students:\n", marks[:3])

# Querying
print("\nStudents with marks > 85:\n", marks[marks > 85])

11. Factorial Using Recursion

# Factorial using recursion

def factorial(n):
    if n == 0 or n == 1:
        return 1
    else:
        return n * factorial(n - 1)

# User input
num = int(input("Enter a number: "))

if num < 0:
    print("Factorial is not defined for negative numbers.")
else:
    result = factorial(num)
    print(f"Factorial of {num} is: {result}")
